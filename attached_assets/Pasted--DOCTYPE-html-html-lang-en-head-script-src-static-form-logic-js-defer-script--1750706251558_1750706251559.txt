document.addEventListener('DOMContentLoaded', function() {
  if (window.READ_ONLY === true || window.READ_ONLY === "true") {
    // 1. Disable all form controls (input, select, textarea)
    document.querySelectorAll('form input, form select, form textarea').forEach(el => {
      // If radio or checkbox, disable it
      if (el.type === 'checkbox' || el.type === 'radio') {
        el.disabled = true;
      }
      // Otherwise, make it readonly and disable (for selects)
      else if (el.tagName === 'SELECT') {
        el.disabled = true;
      }
      else {
        el.readOnly = true;
      }
      el.required = false; // remove required so errors don't trigger
    });

    // 2. Hide ALL "add" and "remove" buttons for dynamic entries
    document.querySelectorAll(
      '.removeAch, .removeBeneficiary, .removeDirect, #addAchBtn, #addBeneficiary, #addDirect'
    ).forEach(btn => btn.style.display = 'none');

    // 3. Disable or hide all Yes/No, toggle, or stepper buttons
    document.querySelectorAll(
      // All nav sidebar buttons EXCEPT for tab navigation
      '.inline-flex button, .inline-flex > button, .nav-btn'
    ).forEach(btn => {
      // Do not disable sidebar nav (just section toggles)
      if (!btn.classList.contains('nav-btn')) {
        btn.disabled = true;
        btn.classList.add('opacity-60', 'pointer-events-none');
      }
    });

    // 4. Hide the submit button(s)
    document.querySelectorAll('button[type="submit"]').forEach(btn => btn.style.display = 'none');

    // 5. Optionally, block any accidental form submission by keyboard
    const form = document.querySelector('form');
    if (form) {
      form.addEventListener('submit', function(e) {
        e.preventDefault();
        return false;
      });
    }

    // 6. Show a lock banner at the top
    const lockBanner = document.createElement('div');
    lockBanner.textContent = 'This account is locked and read-only.';

    const main = document.querySelector('.w-3/4 > .flex') || document.querySelector('.w-3/4');
    if (main) main.prepend(lockBanner);
    else document.body.prepend(lockBanner);
  }
});

// --- Business rules ---
const regTypesRequireBenef = [
  'Roth IRA','SARSEP','SEP IRA','SIMPLE IRA','Traditional IRA',
  'Beneficiary IRA','Beneficiary Roth IRA','Beneficiary SIMPLE IRA',
  'Guardian IRA','Guardian Roth IRA'

const regTypesRequireHolder = [
  'Guardianship','Conservatorship','Education Savings',
  'Minor Custodial','529 Plan','Guardian IRA','Guardian Roth IRA'

// Friendly label lookup
function getFieldLabel(el) {
  if (!el) return '(unnamed)';
  if (el.id) {
    const lbl = document.querySelector(`label[for="${el.id}"]`);
    if (lbl) return lbl.innerText.trim();
  }
  const plbl = el.closest('div')?.querySelector('label');
  if (plbl) return plbl.innerText.trim();
  if (el.placeholder) return el.placeholder;
  if (el.name) return el.name.replace(/_/g,' ').replace(/\b\w/g,c=>c.toUpperCase());
  return '(unnamed)';
}

function showRequiredFieldsSummary() {
  const summary = document.getElementById('requiredFieldsSummary');
  if (!summary) return;

  const acct = document.getElementById('account_type')?.value || '';
  const prog = document.getElementById('program_type')?.value || '';
  const reg  = document.getElementById('registration_type')?.value || '';
  const tod  = document.getElementById('todChoice')?.value || '';

  const required = [], missing = [];

  // 1) Top-level selectors
  ['account_type','program_type','registration_type'].forEach(id => {
    const el = document.getElementById(id);
    const lbl = getFieldLabel(el) || id;
    if (!required.includes(lbl)) required.push(lbl);
    if (!el || !el.value) missing.push(lbl);
  });

  // 2) All HTML5 required fields except those inside .beneficiary-entry
  document.querySelectorAll('input,select,textarea').forEach(el => {
    if (
      el.required &&
      !el.disabled &&
      el.willValidate &&
      !el.closest('.beneficiary-entry')   // <-- skip beneficiary internals
    ) {
      const lbl = getFieldLabel(el);
      if (!required.includes(lbl)) required.push(lbl);
      if (!el.checkValidity() && !missing.includes(lbl)) {
        missing.push(lbl);
      }
    }
  });

// 3) Beneficiaries (#N prefixed)
if (regTypesRequireBenef.includes(reg) || tod === 'Yes') {
  const rows = document.querySelectorAll('.beneficiary-entry');

  // no rows at all ⇒ all three must be present
  if (!rows.length) {
    ['Type','Entity Name','Percentage'].forEach(fld => {
      const lbl = `Beneficiary ${fld}`;
      if (!required.includes(lbl)) required.push(lbl);
      if (!missing.includes(lbl)) missing.push(lbl);
    });
  } else {
    rows.forEach((row, idx) => {
      const rel = row.querySelector('select[name*="[Relationship]"]')?.value;
      const isNonPerson = rel === 'Non-Person';

      // choose which fields to check
      const fields = isNonPerson
        ? ['Type','Entity Name','Percentage']
        : ['First Name','Last Name','Percentage','Relationship','Type'];

      fields.forEach(fld => {
        const lbl = `Beneficiary #${idx+1} ${fld}`;
        if (!required.includes(lbl)) required.push(lbl);

        // locate the input or select by name
        const sel = row.querySelector(`[name*="[${fld}]"]`);
        const val = sel ? sel.value.trim() : '';
        if (!val && !missing.includes(lbl)) {
          missing.push(lbl);
        }
      });
    });
  }
}

  // 4) Additional Holder
  if (regTypesRequireHolder.includes(reg) || acct.toLowerCase().includes('joint')) {
    // First/Last
    ['First Name','Last Name'].forEach(n => {
      const lbl = `Additional Holder ${n}`;
      if (!required.includes(lbl)) required.push(lbl);
      const sel = document.querySelector(`input[name="Additional Holder - ${n}"]`);
      if (!sel || !sel.value.trim()) missing.push(lbl);
    });
    // Phone (any)
    const phoneLbl = 'Additional Holder Phone (any)';
    if (!required.includes(phoneLbl)) required.push(phoneLbl);
    const home = document.querySelector('input[name="Additional Holder - Home Phone"]')?.value.trim();
    const mob  = document.querySelector('input[name="Additional Holder - Mobile Phone"]')?.value.trim();
    const bus  = document.querySelector('input[name="Additional Holder - Business Phone"]')?.value.trim();
    if (!(home||mob||bus)) missing.push(phoneLbl);
    // Employer Address
    const useSame = document.querySelector('input[name="Additional Holder - Use Same Address As Primary"]')?.checked;
    const excl    = document.querySelector('input[name="Additional Holder - Exclude Employer Address"]')?.checked;
    if (!useSame && !excl) {
      ['Employer Address Line 1','Employer City','Employer State','Employer Zip']
      .forEach(n => {
        const lbl = `Additional Holder ${n}`;
        if (!required.includes(lbl)) required.push(lbl);
        const sel = n==='Employer State'
          ? document.querySelector(`select[name="Additional Holder - ${n}"]`)
          : document.querySelector(`input[name="Additional Holder - ${n}"]`);
        if (!sel || !sel.value.trim()) missing.push(lbl);
      });
    }
  }

  // Dedupe & render
  const reqList = Array.from(new Set(required)).sort();
  const missList= Array.from(new Set(missing)).sort();
  if (!reqList.length) { summary.innerHTML=''; return; }

  const items = reqList.map(f =>
    missList.includes(f)
      ? `<li class="text-red-600 font-semibold">${f}</li>`
      : `<li>${f}</li>`
  ).join('');

  summary.innerHTML = `
    <strong>Required fields for this selection:</strong>
    <ul class="list-disc ml-6 mt-1">${items}</ul>
    ${missList.length
      ? `<p class="mt-2 text-red-700"><strong>${missList.length}</strong> required field${missList.length>1?'s':' is'} still missing.</p>`
      : `<p class="mt-2 text-green-700">All required fields are filled out.</p>`
    }
  `;
}

<script>
function forceSummaryUpdateUntilStable(maxAttempts = 20) {
  let attempts = 0;

  function attempt() {
    showRequiredFieldsSummary();

    const message = document.getElementById('requiredFieldsSummary')?.innerText || '';
    const stillMissing = message.includes('required field') || message.includes('still missing');

    if (!stillMissing || attempts >= maxAttempts) return;
    attempts++;
    setTimeout(attempt, 50);
  }

  // Start the loop
  setTimeout(attempt, 50);
}

document.addEventListener('DOMContentLoaded', forceSummaryUpdateUntilStable);

<script>
  document.addEventListener('DOMContentLoaded', () => {
    const MODE         = window.MODE      || 'new';
    const INIT_ACCT    = window.INIT_ACCT || {};
    const container    = document.getElementById('beneficiaryContainer');
    const addBtn       = document.getElementById('addBeneficiary');
    const templateHTML = document.getElementById('beneficiaryTemplate').innerHTML;
    const maxBen       = 6;
    let   benCount     = container.querySelectorAll('.beneficiary-entry').length;
    let   listsData    = {};

    /* ─── utility helpers ─────────────────────────────────────────────── */
    function updateAddBtn() {
      addBtn.style.display = benCount < maxBen ? '' : 'none';
    }
    function bindRemove(btn) {
      btn.addEventListener('click', () => {
        btn.closest('.beneficiary-entry').remove();
        benCount--;
        reindex();
        updateAddBtn();
      });
    }
    function reindex() {
      Array.from(container.querySelectorAll('.beneficiary-entry')).forEach((entry, idx) => {
        entry.dataset.index = idx;
        entry.querySelectorAll('[name]').forEach(el => {
          el.name = el.name.replace(/\[\d+\]/, `[${idx}]`);
          if (el.id) {
            el.id = el.id.replace(/_\d+/, `_${idx}`).replace(/___\d+/, `___${idx}`);
          }
        });
      });
    }
    function addEntry(idx) {
      if (benCount >= maxBen) return;
      const html = templateHTML.replace(/__INDEX__/g, idx);
      container.insertAdjacentHTML('beforeend', html);
      const entry = container.querySelector('.beneficiary-entry:last-child');
      // populate dropdowns
      entry.querySelectorAll('select[data-list-key]').forEach(sel => {
        (listsData[sel.dataset.listKey] || []).forEach(opt => sel.add(new Option(opt, opt)));
      });
      bindRemove(entry.querySelector('.removeBeneficiary'));
      benCount++;
      updateAddBtn();
    }

    /* ─── fetch dropdown lists first ──────────────────────────────────── */
fetch('/lists')
  .then(r => r.json())
  .then(data => listsData = data)
  .finally(() => {
    /* EDIT mode: only add rows for actually-filled beneficiaries */
    if (MODE === 'edit') {
      const benIndices = [...new Set(
        Object.entries(INIT_ACCT)
          // find all Beneficiaries[...] keys that have non-empty values
          .filter(([key, val]) => {
            const m = key.match(/^Beneficiaries\[(\d+)]/);
            return m && String(val).trim() !== '';
          })
          // pull out the numeric index
          .map(([key]) => parseInt(key.match(/^Beneficiaries\[(\d+)]/)[1], 10))
      )]
      // skip index 0 (it’s already in your HTML)
      .filter(i => i > 0)
      .sort((a, b) => a - b);

      // add exactly those rows
      benIndices.forEach(idx => {
        addEntry(idx);
      });

      if (typeof prefillFromRow === 'function') prefillFromRow();
    }

      // initial remove binding & add-btn status
      container.querySelectorAll('.removeBeneficiary').forEach(bindRemove);
      updateAddBtn();
      initNonPersonLogic();
      enforceBeneficiaryFields();
      showRequiredFieldsSummary();
    });

    /* Add-another button */
    addBtn.addEventListener('click', () => addEntry(benCount));
  });
</script>

/* Utility: clear all inputs/selects/textareas within a container */
function clearFields(container) {
  container.querySelectorAll('input, select, textarea').forEach(el => {
    if (el.type === 'checkbox' || el.type === 'radio') {
      el.checked = false;
    } else {
      el.value = '';
      if (el.tagName.toLowerCase() === 'select') el.selectedIndex = 0;
    }
  });
}

<script>
document.addEventListener('DOMContentLoaded', () => {
  const MODE      = window.MODE     || 'new';
  const INIT_ACCT = window.INIT_ACCT|| {};

  // ── Direct/Outside Business setup ─────────────────────────────────────────
  const container = document.getElementById('directContainer');
  const addBtn    = document.getElementById('addDirect');
  const tmplHTML  = document.getElementById('directBusinessTemplate').innerHTML;
  const maxItems  = 5;
  let   count     = container.querySelectorAll('.direct-entry').length;
  let   lists     = {};

  function updateAddBtn() {
    addBtn.style.display = count < maxItems ? '' : 'none';
  }

  function reindex() {
    container.querySelectorAll('.direct-entry').forEach((entry, idx) => {
      entry.dataset.index = idx;
      entry.querySelectorAll('[name]').forEach(el => {
        el.name = el.name.replace(/DirectBusiness\[\d+]/, `DirectBusiness[${idx}]`);
      });
      entry.querySelectorAll('[id]').forEach(el => {
        el.id = el.id
                  .replace(/_\d+$/, `_${idx}`)
                  .replace(/___\d+/, `___${idx}`);
      });
    });
  }

  function bindRemove(btn) {
    btn.addEventListener('click', () => {
      btn.closest('.direct-entry').remove();
      count--;
      reindex();
      updateAddBtn();
    });
  }

  function populateSelects(scope) {
    scope.querySelectorAll('select[data-list-key]').forEach(sel => {
      (lists[sel.dataset.listKey]||[]).forEach(opt => sel.add(new Option(opt,opt)));
      if (sel.dataset.initial) sel.value = sel.dataset.initial;
    });
  }

  function addEntry(idx) {
    if (count >= maxItems) return;
    container.insertAdjacentHTML('beforeend', tmplHTML.replace(/__INDEX__/g, idx));
    const entry = container.querySelector('.direct-entry:last-child');
    populateSelects(entry);
    bindRemove(entry.querySelector('.removeDirect'));
    count++;
    updateAddBtn();
  }

  // fetch your picklists, then wire up DirectBusiness
  fetch('/lists')
    .then(r => r.json()).then(data => lists = data)
    .finally(() => {
      // always populate the “0” row
      populateSelects(container.querySelector('.direct-entry[data-index="0"]'));

      if (MODE === 'edit') {
        // only re-add rows for indices with at least one non-empty value
        const directIdxs = [...new Set(
          Object.entries(INIT_ACCT)
            .filter(([key,val]) => {
              const m = key.match(/^DirectBusiness\[(\d+)]/);
              return m && String(val||'').trim() !== '';
            })
            .map(([key]) => Number(key.match(/^DirectBusiness\[(\d+)]/)[1]))
        )]
          .filter(i => i > 0)
          .sort((a,b)=>a-b);

        directIdxs.forEach(idx => addEntry(idx));

        if (typeof prefillFromRow === 'function') prefillFromRow();
      }

      // finally, bind removes on whatever is present & update add button
      container.querySelectorAll('.removeDirect').forEach(bindRemove);
      updateAddBtn();
    });

  // wire up the +Add button
  addBtn.addEventListener('click', () => addEntry(count));

  // ── ACH rows on edit ───────────────────────────────────────────────────────
  if (MODE === 'edit' && typeof window.addAchEntry === 'function') {
    const achIdxs = [...new Set(
      Object.entries(INIT_ACCT)
        .filter(([key,val]) => {
          const m = key.match(/^ACH\[(\d+)]/);
          return m && String(val||'').trim() !== '';
        })
        .map(([key]) => Number(key.match(/^ACH\[(\d+)]/)[1]))
    )]
      .filter(i => i > 0)
      .sort((a,b)=>a-b);

    achIdxs.forEach(() => window.addAchEntry());
  }
});
</script>

// --- Utility: Clear fields in a container ---
window.clearFields = function(container) {
  if (!container) return;
  container.querySelectorAll('input, select, textarea').forEach(f => {
    if (f.type === 'checkbox' || f.type === 'radio') {
      f.checked = false;
    } else {
      f.value = '';
      if (f.tagName.toLowerCase() === 'select') f.selectedIndex = 0;
    }
  });
};

// --- Sidebar tab navigation ---
window.showSection = function(id, btn) {
  btn.classList.add('visited');
  document.querySelectorAll('form section[id]').forEach(sec => {
    sec.classList.add('hidden');
  });
  document.getElementById(id).classList.remove('hidden');
  document.querySelectorAll('.nav-btn').forEach(b => {
    b.classList.remove('bg-white', 'border-l-4', 'border-blue-800', 'text-blue-800');
    b.classList.add('text-gray-700');
  });
  btn.classList.add('bg-white', 'border-l-4', 'border-blue-800', 'text-blue-800');
  btn.classList.remove('text-gray-700');
};

// --- ACH Row Functions ---
window.toggleAch = function(show, triggeredByUser = false) {
  const container = document.getElementById('achContainer');
  const addBtn    = document.getElementById('addAchBtn');
  const hidden = document.getElementById('achChoice');
  if (hidden) hidden.value = show ? 'Yes' : 'No';

  if (!show) {
    while (container && container.firstChild) {
      container.removeChild(container.firstChild);
    }
    container.classList.add('hidden');
    addBtn.classList.add('hidden');
  } else if (triggeredByUser || window.MODE !== 'edit') {
    while (container && container.firstChild) {
      container.removeChild(container.firstChild);
    }
    const tpl  = document.getElementById('achTemplate').innerHTML;
    const html = tpl.replace(/__INDEX__/g, 0);
    container.insertAdjacentHTML('beforeend', html);
    window.clearFields(container);
    container.classList.remove('hidden');
    addBtn.classList.remove('hidden');
  } else {
    container.classList.remove('hidden');
    addBtn.classList.remove('hidden');
  }

  document.getElementById('achYesBtn')
    .classList.toggle('bg-blue-100', show);
  document.getElementById('achYesBtn')
    .classList.toggle('text-blue-800', show);
  document.getElementById('achNoBtn')
    .classList.toggle('bg-blue-100', !show);
  document.getElementById('achNoBtn')
    .classList.toggle('text-blue-800', !show);
};

window.addAchEntry = function() {
  const container = document.getElementById('achContainer');
  const count     = container.querySelectorAll('.ach-entry').length;
  if (count >= 2) return; // cap at 2 entries

  const tpl  = document.getElementById('achTemplate').innerHTML;
  const html = tpl.replace(/__INDEX__/g, count);
  container.insertAdjacentHTML('beforeend', html);

  if (container.querySelectorAll('.ach-entry').length >= 2) {
    document.getElementById('addAchBtn').classList.add('hidden');
  }
};

      if (achContainer.querySelectorAll('.ach-entry').length < 2) {
        document.getElementById('addAchBtn').classList.remove('hidden');
      }
    });
  }
});
